# Conventional Commits template
# Format:
#   <type>(<scope>): <subject>
#
# Allowed types:
#   feat, fix, docs, style, refactor, perf, test, build, ci, chore, revert
#
# Rules:
# - Keep the subject in imperative mood
# - Prefer a short, specific scope when useful
# - Keep the first line within 72 characters
# - Add a blank line before body/footer when needed
#
# Examples:
# feat(chat): add inspect fallback output
# fix(state): handle missing storage.json
# docs(readme): clarify install steps
#
# Body (optional):
# Explain what changed and why.
#
# Footer (optional):
# BREAKING CHANGE: ...
# Refs: #123
